using Microsoft.AspNetCore.Builder;
using Microsoft.AspNetCore.Http;
using System.Collections.Generic;
using System.Linq;

var builder = WebApplication.CreateBuilder(args);
var app = builder.Build();

var tasks = new List<TodoTask> 
{ 
    new TodoTask { Id = 1, Title = "Submit application before 4 PM", IsCompleted = false },
    new TodoTask { Id = 2, Title = "Review C# basics", IsCompleted = true }
};

app.MapGet("/tasks", () => tasks);

app.MapPost("/tasks", (TodoTask newTask) => 
{
    newTask.Id = tasks.Count > 0 ? tasks.Max(t => t.Id) + 1 : 1;
    tasks.Add(newTask);
    return Results.Created($"/tasks/{newTask.Id}", newTask);
});

app.Run();

class TodoTask 
{
    public int Id { get; set; }
    public string Title { get; set; } = string.Empty;
    public bool IsCompleted { get; set; }
}
