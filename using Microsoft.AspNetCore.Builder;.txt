using Microsoft.AspNetCore.Builder;
using Microsoft.AspNetCore.Http;
using System.Collections.Generic;
using System.Linq;

var builder = WebApplication.CreateBuilder(args);
var app = builder.Build();

var tasks = new List<TodoTask> 
{ 
    new TodoTask { Id = 1, Title = "Submit application before 4 PM", IsCompleted = false },
    new TodoTask { Id = 2, Title = "Review C# basics", IsCompleted = true }
};

// GET all tasks
app.MapGet("/tasks", () => Results.Ok(tasks));

// GET single task by ID (Professional: 404 Error Handling)
app.MapGet("/tasks/{id}", (int id) => 
{
    var task = tasks.FirstOrDefault(t => t.Id == id);
    return task is not null ? Results.Ok(task) : Results.NotFound(new { Error = "Task not found." });
});

// POST new task (Professional: Input Validation)
app.MapPost("/tasks", (TodoTask newTask) => 
{
    if (string.IsNullOrWhiteSpace(newTask.Title))
        return Results.BadRequest(new { Error = "Task title cannot be empty." });

    newTask.Id = tasks.Count > 0 ? tasks.Max(t => t.Id) + 1 : 1;
    tasks.Add(newTask);
    return Results.Created($"/tasks/{newTask.Id}", newTask);
});

// DELETE a task (Professional: Complete CRUD)
app.MapDelete("/tasks/{id}", (int id) => 
{
    var task = tasks.FirstOrDefault(t => t.Id == id);
    if (task is null) 
        return Results.NotFound(new { Error = "Task not found." });
    
    tasks.Remove(task);
    return Results.NoContent();
});

app.Run();

class TodoTask 
{
    public int Id { get; set; }
    public string Title { get; set; } = string.Empty;
    public bool IsCompleted { get; set; }
}
